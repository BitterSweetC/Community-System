\chapter{数据库关键表结构（节选）}

以下 DDL 与项目 \texttt{community-gateway/src/main/resources/schema.sql} 保持一致（节选）：

\begin{lstlisting}[language=SQL, caption=用户、角色、社团与成员表]
CREATE TABLE `t_user` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `password_hash` varchar(255) NOT NULL,
  `real_name` varchar(50) DEFAULT NULL,
  `email` varchar(100) DEFAULT NULL,
  `mobile` varchar(20) DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'ACTIVE',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_username` (`username`)
);

CREATE TABLE `t_role` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `code` varchar(32) NOT NULL,
  `name` varchar(64) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_role_code` (`code`)
);

CREATE TABLE `t_club` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `name` varchar(100) NOT NULL,
  `category` varchar(50) NOT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'PENDING',
  `created_by` bigint NOT NULL,
  `balance` decimal(10,2) DEFAULT '0.00',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_club_name` (`name`)
);

CREATE TABLE `t_member` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `club_id` bigint NOT NULL,
  `user_id` bigint NOT NULL,
  `role_code` varchar(32) NOT NULL DEFAULT 'MEMBER',
  `status` varchar(20) NOT NULL DEFAULT 'ACTIVE',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_club_user` (`club_id`,`user_id`)
);
\end{lstlisting}

\begin{lstlisting}[language=SQL, caption=活动、招新、通知相关表]
CREATE TABLE `t_activity` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `club_id` bigint NOT NULL,
  `title` varchar(200) NOT NULL,
  `start_time` datetime(6) NOT NULL,
  `end_time` datetime(6) NOT NULL,
  `checkin_code` varchar(20) DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'DRAFT',
  PRIMARY KEY (`id`)
);

CREATE TABLE `t_recruit_batch` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `club_id` bigint NOT NULL,
  `title` varchar(100) NOT NULL,
  `start_time` datetime(6) NOT NULL,
  `end_time` datetime(6) NOT NULL,
  `quota` int DEFAULT NULL,
  `status` varchar(20) NOT NULL DEFAULT 'DRAFT',
  PRIMARY KEY (`id`)
);

CREATE TABLE `t_recruit_application` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `batch_id` bigint NOT NULL,
  `user_id` bigint NOT NULL,
  `apply_data` json NOT NULL,
  `first_review_status` varchar(20) DEFAULT 'PENDING',
  `final_review_status` varchar(20) DEFAULT 'PENDING',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_batch_user` (`batch_id`,`user_id`)
);

CREATE TABLE `t_notification` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `user_id` bigint NOT NULL,
  `title` varchar(200) NOT NULL,
  `content` text NOT NULL,
  `type` varchar(20) NOT NULL,
  `is_read` bit(1) NOT NULL DEFAULT b'0',
  PRIMARY KEY (`id`)
);
\end{lstlisting}

\chapter{部署与运行说明}

\begin{enumerate}
  \item 准备环境变量（数据库、Redis、RabbitMQ、JWT、邮件与 OSS 配置）；
  \item 使用 Docker Compose 启动基础服务：\texttt{mysql}、\texttt{redis}、\texttt{rabbitmq}、\texttt{rag-service}；
  \item 启动后端应用容器 \texttt{community-app}；
  \item 启动前端容器 \texttt{community-frontend}，由 Nginx 反向代理 \texttt{/api} 到后端；
  \item 使用 Prometheus 与 Grafana 观察运行指标并进行运维监控。
\end{enumerate}

\chapter{AI 辅助写作与核验说明}

本文在选题梳理、结构规划、语言润色、引用核对与格式检查阶段使用了 AI 辅助工具。AI 参与范围主要包括：论文结构建议生成、章节修改意见整理、参考文献存在性核验流程支持、摘要与结论口径一致性检查，以及交付阶段的格式与材料清单整理。

需要强调的是，本文的研究选题、系统实现、测试执行、结果判断、结论取舍与最终文字责任均由作者本人承担。凡涉及系统功能、实验数据、性能结果、评估边界与学术结论的内容，均以项目代码、测试工件、人工核验结果及作者最终审定为准，而非直接采信 AI 自动生成内容。

对 AI 输出的使用遵循以下原则：
\begin{enumerate}
  \item 不将 AI 生成内容未经核验直接作为事实性证据写入正文；
  \item 对引用、数据与研究结论采用人工复核与工件对照；
  \item 对无法独立验证的 AI 输出不纳入正式结论；
  \item 由作者对全文准确性、完整性与学术诚信承担最终责任。
\end{enumerate}
